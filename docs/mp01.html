# Task 1
#' Acquire Netflix What We Watched Reports
#' 
#' This function will acquire and standardize all data for MP#01
#' from Netflix. Because Netflix hosts the data files behind a CDN, we
#' are accessing the release pages and automatically extracting links to the
#' relevant files (with CDN hash). 
#' 
#' After import, files are stacked vertically and some basic cleaning is
#' performed. 
load_netflix_data <- function(){
  # Note to students: Make sure you have installed the tidyverse and fs packages
  #                   before beginning this assignment. If you aren't sure,
  #                   simply run `library(tidyverse)` at your R Console and, if
  #                   you don't see an error message, `tidyverse` has been 
  #                   installed. (Same for `fs`) If you still need to install
  #                   either package, use the `install.packages` function as 
  #                   demonstrated in class.
  library(tidyverse)
  library(rvest)
  library(readxl)
  library(fs)
  
  data_dir <- path("data", "mp01")
    
  if(!dir_exists(data_dir)){
    dir_create(data_dir, showWarnings=FALSE, recurse=TRUE)
  }
  
  NETFLIX_URLS <- list(
    `2023H1` = "https://about.netflix.com/news/what-we-watched-a-netflix-engagement-report",
    `2023H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2023",
    `2024H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2024",
    `2024H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2024",
    `2025H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2025",
    `2025H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2025",
    `2026H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2026"
  )
  
  download_if_needed <- function(url, local){
    if(!file_exists(local)){
      download.file(url, local, quiet = TRUE, mode = "wb")
    }
    
    return(local)
  }
    
  NETFLIX_URLS |> 
    map(read_html) |> 
    map(html_element, xpath = "//a[contains(@href, '.xlsx')]") |> 
    map_chr(html_attr, "href") |> 
    enframe("period", "url") |> 
   mutate(path = path(data_dir, paste0(period, ".xlsx")), 
          path = map2(url, path, download_if_needed), 
          data = map(path, read_excel, skip=3, na="*"), 
          data = map(data, \(x) x |> mutate(`Release Date` = as.Date(`Release Date`)))) |>
    unnest(data) |> 
    select(-url, -path) |> 
    mutate(year=str_sub(period, 1, 4), 
           half=str_sub(period, start=6), 
           period_start = make_date(year, if_else(half==1, 1, 7), 1), 
           period_end = make_date(year, if_else(half==1, 7, 12), if_else(half==1, 30, 31))) |> 
    filter(!is.na(`Hours Viewed`)) |> 
    mutate(live_event = !is.na(Runtime) & str_detect(Runtime, "\\d+:\\d+\\*"), 
           clean_time = str_remove(Runtime, fixed("*")),
           Runtime = 60 * hour(hm(clean_time)) + minute(hm(clean_time))) |>
    select(-clean_time) |>
    mutate(Runtime = coalesce(Runtime, mean(Runtime, na.rm=TRUE)), 
           .by=Title, 
           Title = stringi::stri_trans_general(Title, "Latin-ASCII"),
           Title = str_remove(Title, " // .*"), 
           Title = str_remove(Title, "[^[:ascii:]].*"), 
           Title = str_trim(Title)) |>
    filter_out(Title == "") |>
    select(year, 
           period, 
           period_start, 
           period_end, 
           Title, 
           -half, 
           Runtime, 
           Views, 
           `Hours Viewed`, 
           `Release Date`,
           `Available Globally?`, 
           live_event) |>
           
# Data Cleaning and Preparation

# Task 2 Fixing column titles for easy data manipulation
# Note: run_time is measured in minutes
        
    rename(
           title = Title,
           run_time = Runtime,
           views = Views,
           hours_viewed = `Hours Viewed`,
           release_date = `Release Date`,
           available_globally = `Available Globally?`) |>
           
# Task 3 Fixing column types in order to properly perform future analysis. 

    mutate(
           year = as.integer(year),
           available_globally = available_globally == "Yes")
           

}

Question 1: What was the top title (by total hours viewed) in each period and what fraction of total viewership did it capture?

# Creating a column for estimated views. This added column provide us values for any observations that has a missing value for their views column.
NETFLIX_DATA_with_view_estimates <- NETFLIX_DATA |>
  mutate(
    estimated_views = if_else(
      is.na(views) & !is.na(run_time) & run_time > 0,
      hours_viewed / (run_time / 60),
      NA_real_
    )
  )

# Adding a view count column for the calculations. This column will use the values provided to us by Netflix, otherwise it will use the values from our estimated_views column.
NETFLIX_DATA_with_view_estimates <- NETFLIX_DATA_with_view_estimates |>
  mutate(
    views_to_use = coalesce(views, estimated_views)
  )
  
# Finding top title by hours viewed and calculating its fraction of total viewership
top_titles <- NETFLIX_DATA_with_view_estimates |>
  group_by(period) |>
  mutate(
    total_views = sum(views_to_use, na.rm = TRUE)
  ) |>
  slice_max(
    order_by = hours_viewed,
    n = 1,
    with_ties = FALSE
  ) |>
  mutate(
    viewership_share = views_to_use / total_views
  ) |>
  ungroup() |>
  select(
    period,
    title,
    hours_viewed,
    views,
    estimated_views,
    views_to_use,
    total_views,
    viewership_share
  )

The table above shows the proportion of viewership associated with the top title (ranked by hours viewed) of each reporting period. The “Recorded Views” column consist of values that Netflix gave us. They are noted as “implied views” (computed by dividing Hours Viewed by run time) assuming that that all viewers watched the entirety of the said title. We use the same methodology to fill in the missing values for this column and labeled them as “Estimated Views”. When performing the calculations to obtain the percentage, We use the estimated views only if the value under recorded views is missing. Otherwise we use the values that Netflix gave us.

The table gives us information on how many views a “typical” popular show will accumulate, as well as the percentage of viewership it would hold for each reporting period. Starting from 2025 we see a “jump” in hours viewed and viewership. Hours viewed is moving closer to the billion range and viewership are now in the 100 million range.

Corporate Strategy Memo

Company: Netflix
To: Leadership
From: Data Science Department
Date 10/17/2026

The purpose of this memo is to create a strategy in how Netflix’s original content should be invested for the 2027 year. After doing several analyses based off our What We Watch (WWW) data, we came to a conclusion on several strategies that Netflix can take on for the upcoming year.

Strategy 1
Investing in more shows than movies. One of the analysis that we’ve done compared the hours viewed regarding shows vs movies. We found that across all 7 periods in which the WWW data was available, users spent a relatively large amount of time watching shows rather than movies. For each period the fraction of viewing time occupied by shows is more than 99 percent. Intuitively this makes sense being that the run time of a show’s season is longer than a standard movie. If both a popular movie and a show have the same number of viewers, the aggregated view time would be much greater for the show than for the movie.

Furthermore in a separate analysis that was done, regarding consistent high viewership, the top five titles ranked by average viewership across all reporting periods happens to be all shows. Thus further supporting strategy 1.

Finally what type of shows do we want to invest in? According to our analysis, children show are ranked highest regarding consistent high viewership. Peppa Pig: Season 6, Peppa Pig: Season 3, and Peppa Pig: Season 4 are ranked top 3, while Gabby’s Dollhouse: Season 1 and Masha and the Bear: Season 5 follows after.

All in all, investing in shows (particularly children shows) will create not only longer engagement (hours viewed) but also consistent engagement across multiple periods.

Strategy 2
Investing in more contents that are globally licensed. We did an additional analysis where we looked at aggregated viewing hours of titles that were always globally available to titles that were never.

According to our analysis, starting from 2023H2, the ratio between globally available and non-globally available contents have not changed. For each reporting period, we had fewer globally available contents. However, the average viewing hours and average viewership were higher when compared to non-globally available contents.

We also notice that in some periods (i.e. 2024H2 and 2026H1) the total hours viewed for non-globally licensed content was higher when compared to globally licensed content, despite having lower averages. This can be counter intuitive being that we would expect contents available to a larger audience having more aggregated viewing hours than contents that are not.

Nevertheless, five of the seven reporting periods do align with our expectations. Despite those 2 periods having lower aggregated viewing hours all the periods still have higher averages for globally available events. Thus, supporting our claim that for the average content having a wider audience correlates with more views and viewing hours. We believe increasing the ratio of globally available contents to non-globally available contents is a worthwhile strategy to take on.

Strategy 3
Putting more focus on expanding the catalog rather than renewing contents. In one of our earlier analysis, we looked at the change in the size of Netflix’s catalog over time. We went further into the analysis to see if hours viewed and number of views (viewership) also reflects a similar trend to what we were seeing.

Note that 2023H1 has “non applicable” values due to having no prior period to measure against. Also, 2023H1 has a large number of missing values regarding its run time, and large number of titles. We mentioned in our initial analysis that this could be due to the quality of our original data set. We included it and the percentage change calculated for 2023H2 in our table for informational purpose, but the actual observations starts from 2024H1.

From our initial analysis we noted that there is a consistent growth of Netflix’s catalog regarding number of titles and total run time. Interestingly, hours viewed and viewership, does not reflect a similar trend. We noticed that both metrics decreased in the second half of the year and picked up again in the following year. However the decrease is extremely minimal (less than one percent) that it does not influence our stance on strategy 3. Hours viewed and viewership does show a consistent increase on a yearly basis. Although it may not be a driver, the increase in the catalogs size does go along with the increase regarding hours viewed and viewership.

We believe incorporating these 3 core strategies:

  • Investing in more shows than movies
  • Investing in more globally licensed contents
  • Putting more focus on expanding our catalog than renewing contents

will provide a positive impact on business in the year 2027.
